
# 1. Доступ к /api/data без токена -> 401
$ curl -i http://127.0.0.1:5055/api/data
HTTP/1.1 401 UNAUTHORIZED
{"error":"Требуется токен в заголовке Authorization: Bearer <token>"}

# 2. Логин с неверным паролем -> 401
$ curl -X POST http://127.0.0.1:5055/auth/login -d '{"username":"student","password":"wrong"}'
{"error":"Неверный логин или пароль"}


# 3. Логин с верным паролем -> 200 и JWT
$ curl -X POST http://127.0.0.1:5055/auth/login -d '{"username":"student","password":"***"}'
{"access_token":"eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIi...","token_type":"Bearer","expires_in":900}

# 4. Создание поста с XSS-нагрузкой (свой метод POST /api/posts) -> 201, ответ экранирован
$ curl -X POST http://127.0.0.1:5055/api/posts -H 'Authorization: Bearer $TOKEN' -d '{"title":"<script>alert(1)</script>","body":"hello"}'
{"author":"student","body":"hello","created_at":"2026-11-04 21:39:28","id":1,"title":"&lt;script&gt;alert(1)&lt;/script&gt;"}

# 5. GET /api/data с токеном -> 200
$ curl http://127.0.0.1:5055/api/data -H 'Authorization: Bearer $TOKEN'
{"count":1,"items":[{"author":"student","body":"hello","created_at":"2026-11-04 21:39:28","id":1,"title":"&lt;script&gt;alert(1)&lt;/script&gt;"}]}

# 6. Попытка SQL-инъекции в параметре q -> 0 результатов, ошибок нет
$ curl "http://127.0.0.1:5055/api/data?q=' OR '1'='1" -H 'Authorization: Bearer $TOKEN'
{"count":0,"items":[]}

# 7. Поддельный токен -> 401
$ curl http://127.0.0.1:5055/api/data -H 'Authorization: Bearer eyJhbGciOiJub25lIn0.e30.'
{"error":"Недействительный или просроченный токен"}
